// src/entities/repair-order.entity.ts
import { Entity, PrimaryGeneratedColumn, Column, CreateDateColumn, UpdateDateColumn } from 'typeorm';

export enum RepairStatus {
  RECEIVED = 'RECEIVED',         // Recibido en tienda
  IN_REPAIR = 'IN_REPAIR',       // En banco de trabajo / reparación
  WAITING_PARTS = 'WAITING_PARTS', // Esperando repuestos del inventario o proveedor
  READY = 'READY',               // Listo para entregar al cliente
  DELIVERED = 'DELIVERED',       // Entregado y cerrado
  CANCELLED = 'CANCELLED',       // Cancelado
}

@Entity('repair_orders')
export class RepairOrder {
  @PrimaryGeneratedColumn('uuid')
  id: string;

  @Column({ type: 'varchar', length: 50, unique: true })
  ticketNumber: string; // Ej: TALLER-2026-0001

  @Column({ type: 'uuid' })
  clientId: string; // Cliente dueño del equipo

  @Column({ type: 'uuid' })
  branchId: string; // Sucursal donde se recibe

  @Column({ type: 'uuid', nullable: true })
  technicianId: string; // Empleado técnico asignado

  @Column({ type: 'varchar', length: 100 })
  deviceType: string; // Ej: "Laptop Dell Inspiron", "iPhone 14 Pro"

  @Column({ type: 'varchar', length: 100, nullable: true })
  serialNumber: string; // Serial único o MAC Address (clave para tecnología y electrónicos)

  @Column({ type: 'text' })
  problemDescription: string; // Falla reportada por el cliente

  @Column({ type: 'text', nullable: true })
  aestheticCondition: string; // Notas físicas iniciales (ej: "Rayas leves en pantalla, sin cargador")

  @Column({ type: 'text', nullable: true })
  technicalNotes: string; // Diagnóstico o notas del técnico durante el proceso

  @Column({ 
    type: 'enum', 
    enum: RepairStatus, 
    default: RepairStatus.RECEIVED 
  })
  status: RepairStatus;

  @Column({ type: 'jsonb', default: [] })
  partsUsed: Array<{
    productId: string;
    productName: string;
    quantity: number;
    unitPrice: number;
  }>; // Piezas o repuestos consumidos del inventario

  @Column({ type: 'decimal', precision: 10, scale: 2, default: 0.00 })
  laborCost: number; // Costo de la mano de obra técnica

  @Column({ type: 'decimal', precision: 10, scale: 2, default: 0.00 })
  totalAmount: number; // Total general (Repuestos + Mano de obra)

  @CreateDateColumn()
  createdAt: Date;

  @UpdateDateColumn()
  updatedAt: Date;
}

// src/entities/repair-order.entity.ts
import { Entity, PrimaryGeneratedColumn, Column, CreateDateColumn, UpdateDateColumn } from 'typeorm';

export enum RepairStatus {
  DIAGNOSIS = 'DIAGNOSIS',         // En diagnóstico / revisión
  WAITING_PARTS = 'WAITING_PARTS', // Esperando piezas o repuestos
  IN_PROGRESS = 'IN_PROGRESS',     // En reparación activa
  READY = 'READY',                 // Listo para entregar
  DELIVERED = 'DELIVERED',         // Entregado al cliente
  CANCELLED = 'CANCELLED',         // Cancelado
}

@Entity('repair_orders')
export class RepairOrder {
  @PrimaryGeneratedColumn('uuid')
  id: string;

  @Column({ type: 'uuid' })
  companyId: string;

  @Column({ type: 'uuid' })
  clientId: string;

  @Column({ type: 'uuid', nullable: true })
  technicianId: string; // Vinculado al módulo de empleados que creamos antes

  @Column({ type: 'varchar', length: 150 })
  deviceModel: string; // Ej: "iPhone 13 Pro", "Laptop Lenovo ThinkPad"

  @Column({ type: 'varchar', length: 100, unique: true, nullable: true })
  serialNumberOrMac: string; // Número de serie o MAC Address para control y garantías

  @Column({ type: 'text' })
  problemDescription: string; // Descripción del fallo reportado por el cliente

  @Column({ type: 'text', nullable: true })
  technicalDiagnosis: string; // Diagnóstico realizado por el técnico

  @Column({ type: 'decimal', precision: 12, scale: 2, default: 0 })
  estimatedCost: number; // Costo estimado de la reparación

  @Column({ type: 'decimal', precision: 12, scale: 2, default: 0 })
  finalCost: number; // Costo final de la reparación

  @Column({ 
    type: 'enum', 
    enum: RepairStatus, 
    default: RepairStatus.DIAGNOSIS 
  })
  status: RepairStatus;

  @CreateDateColumn()
  createdAt: Date;

  @UpdateDateColumn()
  updatedAt: Date;
}
